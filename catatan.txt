-----generate kode user

CREATE OR REPLACE FUNCTION generate_kode_user(
    nama VARCHAR, 
    tempat_lahir VARCHAR, 
    tgl_lahir DATE
) 
RETURNS VARCHAR AS $$
BEGIN
    -- Contoh format: USR-NAMA-TAHUN-RANDOM (Silakan modifikasi sesuai kebutuhan)
    RETURN 'USR-' || 
           UPPER(LEFT(nama, 3)) || '-' || 
           EXTRACT(YEAR FROM tgl_lahir) || '-' || 
           LPAD(FLOOR(RANDOM() * 1000)::TEXT, 3, '0');
END;
$$ LANGUAGE plpgsql;
